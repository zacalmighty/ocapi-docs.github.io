Megamenu ​

Navigation megamenu slots

Template Source ​

html
{{!-- MEGAMENU - Navigation Megamenu Slot Template --}} {{!-- Used for: MEGAMENU SLOT 1, MEGAMENU
SLOT 2 --}} {{!-- JSON fields: copy, destination.id, destination.ipid, action, altText --}}

<div class="megamenu-slot" data-asset-id="{{assetId}}" data-asset-type="{{assetType}}">
  {{#each items}} {{#unless (eq action "REMOVE")}}
  <div class="megamenu-item" data-order="{{order}}" data-action="{{action}}">
    {{#if destination.id}}
    <a href="{{sfccUrl destination.id destination.ipid}}" class="megamenu-link">
      <span class="megamenu-text">{{copy}}</span>
      {{#if altText}}
      <span class="megamenu-alt">{{altText}}</span>
      {{/if}}
    </a>
    {{else}}
    <span class="megamenu-text">{{copy}}</span>
    {{/if}}
  </div>
  {{/unless}} {{/each}}
</div>

Usage ​

This template is resolved via resolveTemplate(assetType) in src/config/slots.config.js.

javascript
const { resolveTemplate } = require('./config/slots.config');
const template = resolveTemplate('MEGAMENU');
// Returns: "megamenu/template"
