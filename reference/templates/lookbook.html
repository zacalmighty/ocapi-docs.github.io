Lookbook ​

Image gallery and carousel components

Template Source ​

html
{{!-- LOOKBOOK - Image Gallery/Carousel Template --}} {{!-- Used for: IMAGE SLOT 1-10, LOOKBOOK LP,
CAROUSEL --}} {{!-- JSON fields: copy, destination.id, destination.ipid, action, altText --}}

<div class="lookbook-container" data-asset-id="{{assetId}}" data-asset-type="{{assetType}}">
  <div class="lookbook-gallery">
    {{#each items}} {{#unless (eq action "REMOVE")}}
    <div class="lookbook-item" data-order="{{order}}" data-action="{{action}}">
      {{#if destination.id}}
      <a href="{{sfccUrl destination.id destination.ipid}}" class="lookbook-link">
        <div class="lookbook-image-wrapper">
          {{!-- Image would be rendered by SFCC based on asset configuration --}}
          <div class="lookbook-placeholder">
            {{#if altText}}
            <span class="lookbook-alt">{{altText}}</span>
            {{else if copy}}
            <span class="lookbook-label">{{copy}}</span>
            {{/if}}
          </div>
        </div>
        {{#if copy}}
        <p class="lookbook-caption">{{copy}}</p>
        {{/if}}
      </a>
      {{else}}
      <div class="lookbook-image-wrapper">
        <div class="lookbook-placeholder">
          {{#if altText}}
          <span class="lookbook-alt">{{altText}}</span>
          {{else if copy}}
          <span class="lookbook-label">{{copy}}</span>
          {{/if}}
        </div>
      </div>
      {{#if copy}}
      <p class="lookbook-caption">{{copy}}</p>
      {{/if}} {{/if}}
    </div>
    {{/unless}} {{/each}}
  </div>
</div>

Usage ​

This template is resolved via resolveTemplate(assetType) in src/config/slots.config.js.

javascript
const { resolveTemplate } = require('./config/slots.config');
const template = resolveTemplate('LOOKBOOK');
// Returns: "lookbook/template"
