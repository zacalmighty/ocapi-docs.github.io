PLP Header ​

Product listing page header banners

Template Source ​

html
{{!-- PLP_HEADER - Product Listing Page Header Template --}} {{!-- Auto-generated from JSON data
--}} {{!-- JSON fields used: copy, destination.id, destination.ipid, action --}} {{!-- Optional:
backgroundColor, textColor, smallText, couponText, disclaimer, ctas --}}

<!-- If necessary, update path to stylesheet -->
<link rel="stylesheet" href="webassets/headers/2023/g3.plp_header_V6.css?$staticlink$" />

<div
  id="default-plp-header"
  class="g3-plp-header"
  style="background-color: {{#if backgroundColor}}{{backgroundColor}}{{else}}#f8cacc{{/if}}; color: {{#if textColor}}{{textColor}}{{else}}#000{{/if}}"
>
  <div class="g3-plp-header__content">
    <div class="g3-plp-header__txt g3-plp-header__txt--left">
      <!-- Desktop far left / Mobile top center-->
      {{#if leftText}}
      <p>{{leftText}}</p>
      {{/if}}
    </div>

    <!-- Center Section -->
    <div class="g3-plp-header__txt g3-plp-header__txt--center">
      <div>
        <h2>{{#if smallText}}<small>{{smallText}}</small><br />{{/if}}{{{copy}}}<sup>*</sup></h2>
        {{#if couponText}}<span class="g3-coupon" style="text-transform: none">{{couponText}}</span
        >{{/if}}
      </div>
    </div>

    <!-- OPTIONAL CTA(s) Desktop far right / Mobile centered under promos -->
    {{#if ctas}}
    <div class="g3-plp-header__txt g3-plp-header__txt--right">
      {{#each ctas}}
      <a href="{{sfccUrl this.destination.id this.destination.ipid}}" class="g3-cta--ul"
        >{{this.text}}</a
      >
      {{/each}}
    </div>
    {{else if destination.id}}
    <div class="g3-plp-header__txt g3-plp-header__txt--right">
      <a href="{{sfccUrl destination.id destination.ipid}}" class="g3-cta--ul">Shop Now</a>
    </div>
    {{/if}}
  </div>

  <!-- Bottom Disclaimer -->
  <span class="g3-disclaimer--fixed g3-text--center">
    {{#if disclaimer}}{{disclaimer}}{{else}}Online only. Select styles. Non-combinable.{{/if}}
    <a style="text-decoration: underline" href="" class="global-promo-modal-btn">Details</a>.
  </span>
</div>

Usage ​

This template is resolved via resolveTemplate(assetType) in src/config/slots.config.js.

javascript
const { resolveTemplate } = require('./config/slots.config');
const template = resolveTemplate('PLP HEADER');
// Returns: "plp-header/template"
