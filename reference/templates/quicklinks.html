Quicklinks ​

Quick navigation links and mobile pills

Template Source ​

html
{{!-- QUICKLINKS - Navigation Pills Template --}} {{!-- Used for: QUICKLINKS 1-6, MOBILE PILL 1-14
--}} {{!-- JSON fields: copy, destination.id, destination.ipid, action --}}

<div class="quicklinks-container" data-asset-id="{{assetId}}" data-asset-type="{{assetType}}">
  <nav class="quicklinks-nav" aria-label="Quick links">
    {{#each items}} {{#unless (eq action "REMOVE")}} {{#if destination.id}}
    <a
      href="{{sfccUrl destination.id destination.ipid}}"
      class="quicklink-pill"
      data-order="{{order}}"
      data-action="{{action}}"
    >
      {{copy}}
    </a>
    {{else}}
    <span class="quicklink-pill" data-order="{{order}}">{{copy}}</span>
    {{/if}} {{/unless}} {{/each}}
  </nav>
</div>

Usage ​

This template is resolved via resolveTemplate(assetType) in src/config/slots.config.js.

javascript
const { resolveTemplate } = require('./config/slots.config');
const template = resolveTemplate('QUICKLINKS');
// Returns: "quicklinks/template"
