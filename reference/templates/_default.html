Default ​

Fallback template for unmapped asset types

Template Source ​

html
{{!-- Default Template - Used for asset types without specific templates --}}

<div class="content-asset" data-asset-id="{{assetId}}" data-asset-type="{{assetType}}">
  {{#if items}} {{#each items}}
  <div class="content-item" data-order="{{order}}" data-action="{{action}}">
    {{#if copy}}
    <div class="copy">{{copy}}</div>
    {{/if}} {{#if destination}}
    <a href="{{destination.id}}" class="cta">
      {{#if destination.copy}}{{destination.copy}}{{else}}Shop Now{{/if}}
    </a>
    {{/if}} {{#if altText}}
    <span class="alt-text">{{altText}}</span>
    {{/if}}
  </div>
  {{/each}} {{else}}
  <div class="content-placeholder">
    <p>{{assetType}} - {{zone}}</p>
  </div>
  {{/if}}
</div>

Usage ​

This template is resolved via resolveTemplate(assetType) in src/config/slots.config.js.

javascript
const { resolveTemplate } = require('./config/slots.config');
const template = resolveTemplate('DEFAULT');
// Returns: "_default"
