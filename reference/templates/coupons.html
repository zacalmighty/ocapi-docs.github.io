Coupons ​

Coupon code pages with promo details

Template Source ​

html
{{!-- COUPONS - Current Offers Page Template --}}
{{!-- Auto-generated from JSON data. Each item becomes a promo card. --}}
{{!-- JSON fields: copy (heading\ndetails format), destination.id, action --}}
{{!-- copy first line = heading, remaining lines = details --}}

<div class="row promo-offers">
  <div class="col-12 px-0 coupon-heading-column">
    <div class="coupons-header">
      <p class="h3">CURRENT OFFERS</p>
    </div>
    <div class="cd-sub-header">
      <p>Tap<span class="sv-icon sv-black-tick mx-1"></span>below to apply the offer to cart.</p>
    </div>
  </div>

  {{#each items}}
    {{#unless (eq action "REMOVE")}}
    <!------------------------------------------------------------------------>
    <div class="col-lg-4 col-md-12 px-0 promo-grid">
    <div class="promotion-box implicit" style="background: #ffe9eb">
      <div class="heading-container">
        <div class="promotion-heading">
          <p class="h3">{{firstLine copy}}</p>
        </div>
        <div class="promotion-sub-heading">
          <p>{{subheading}}</p>
        </div>
      </div>
      <div class="promo-content">
        <div class="promotion-code">
          <p><span>{{#if code}}Code: {{code}}{{else}}NO CODE NEEDED{{/if}}</span></p>
        </div>
        <div class="coupon-expiry">
          <p>{{expiry}}</p>
        </div>
        <div class="promo-disclaimer">
          <p>
            Exclusions apply<span
              class="promo-see-details global-promo-modal-btn"
              data-component="promo-see-details"
            >
              - <u>Details</u>.</span
            >
          </p>
        </div>
      </div>
      <div class="btns-container">
        {{#if destination.id}}
        <a class="btn btn-outline-primary shop-now" href="{{sfccUrl destination.id ipid}}"
          >Shop Now</a
        >
        {{/if}}
        <div class="cd-buttons">
          {{#if code}}
          <button
            type="button"
            data-code="{{code}}"
            data-component="add-coupon"
            class="btn btn-primary promo-cd-add-coupon"
          >
            <span class="sv-icon sv-black-tick mx-1 js-cd-icon"></span
            ><span data-component="add-coupon-cta" class="add-coupon-cta">{{code}}</span>
          </button>
          {{else}}
          <button type="button" class="btn btn-primary promo-cd-add-coupon">
            <span class="sv-icon sv-black-tick mx-1 js-sv-icon"></span>NO CODE NEEDED
          </button>
          {{/if}}
          <span class="promo-see-details global-promo-modal-btn" data-component="promo-see-details"
            >Details</span
          >
        </div>
      </div>
    </div>
  </div>
    {{/unless}}
  {{/each}}
</div>

Usage ​

This template is resolved via resolveTemplate(assetType) in src/config/slots.config.js.

javascript
const { resolveTemplate } = require('./config/slots.config');
const template = resolveTemplate('COUPONS');
// Returns: "coupons/template"
